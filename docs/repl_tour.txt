# ---------- functions, closures, recursion ----------
:square(n)
    n * n
.
square(9)
:apply(f n)
    f(n)
.
apply(square 6)
:at-most(limit)
    :ok(v)
        v <= limit
    .
    ok
.
le3 : at-most(3)
le3(2)
le3(9)
:sum(n acc)
    if
        n == 0 >> acc
        else >> sum((n - 1) (acc + n))
    .
.
sum(100000 0)

# ---------- flow & dot-flow ----------
:dbl(x)
    x * 2
.
:inc(x)
    x + 1
.
10 >> dbl() >> inc()
[1 2 3 4].size

# ---------- conditionals ----------
:grade(x)
    if
        x > 10 >> "big"
        x > 5 >> "medium"
        else >> "small"
    .
.
grade(20)
grade(7)
grade(1)
if any
    3 > 0 >> print("positive")
    3 > 10 >> print("huge")
    else >> print("none")
.
msg :
    if
        1 < 2 >> "yes"
        else >> "no"
    .
msg
